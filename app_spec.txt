  <project_name>New-TP-Opp</project_name>

  <overview>
    A web application that extracts financial data from Luxembourg annual accounts (PDFs) and produces transfer pricing analysis summaries. Upload a PDF, get structured data + TP opportunity flags. Automates 2-3 hours of manual financial statement review into a 30-second upload for transfer pricing professionals.

  <extraction_prompt>
    <location>See TP_EXTRACTOR_PROMPT.md for full prompt specification</location>
    <summary>
      System prompt establishes Claude as Luxembourg TP specialist.
      User prompt contains JSON schema to follow.
      PDF attached as document.
      Response is pure JSON (no markdown).
      Critical rules:
      - Extract EXACT figures from document
      - Use null for missing data (not zero)
      - Distinguish between "0" (disclosed) and null (not found)
      - Extract current year AND previous year
      - Include page references for major data points
      - All values in EUR
    </summary>
  </extraction_prompt>

  <tp_analysis_logic>
    <location>See TP_ANALYSIS_CONFIG.md for configurable rules</location>
    <debt_equity_analysis>
      - Calculate: debt funding participations / total equity
      - No statutory safe harbour in Luxembourg
      - Thresholds based on market practice:
        - D/E &lt;= 1.5x (&lt;= 60% debt): No flag
        - D/E 1.5x-5.67x (60-85% debt): Low attention
        - D/E 5.67x-10x (85-90% debt): Medium attention
        - D/E &gt; 10x (&gt; 90% debt): High attention
      - Tax risks: deduction denial, recharacterisation to dividend
    </debt_equity_analysis>
    <spread_analysis>
      - Implied lending rate = IC interest income / IC loans granted
      - Implied borrowing rate = IC interest expense / IC loans received
      - Spread = lending rate - borrowing rate
      - Flags:
        - Zero spread: &lt; 10 bps
        - Low spread: &lt; 25 bps
        - Negative spread: borrowing &gt; lending (unusual)
      - Caveats: implied rates are estimates, currency/timing effects
    </spread_analysis>
    <scoring>
      - Score A (High): Zero/negative spread, OR IC &gt; EUR 100M, OR D/E &gt; 10x, OR 2+ high flags
      - Score B (Medium): IC &gt; EUR 20M, OR cash pooling, OR material services, OR D/E 5.67-10x
      - Score C (Low): Default if not A or B
    </scoring>
  </tp_analysis_logic>

  <json_response_schema>
    <structure>
      metadata:
        - company_name, rcs_number, address
        - financial_year_start, financial_year_end
        - currency, account_type
        - extraction_confidence, extraction_notes

      balance_sheet:
        - assets (fixed_assets, current_assets, prepayments, total)
        - liabilities (capital_and_reserves, provisions, creditors, total)
        - IC items clearly identified within each section

      profit_and_loss:
        - All P&L line items with current/previous year
        - IC income and expense separated

      notes_extraction:
        - shares_in_affiliated_details (array with counterparty info)
        - loans_to_affiliated_details (counterparty, currency, rate, maturity)
        - loans_from_affiliated_details (counterparty, currency, rate, maturity)
        - related_party_transactions
        - cash_pooling (exists, counterparty, balances)
        - employees (FTE count)
        - off_balance_sheet_commitments

      entity_classification:
        - primary_type (operational/holding/financing/ip_holding/mixed)
        - activities_description
        - substance_indicators

      tp_analysis:
        - ic_financing (loans, interest, rates, spread, flags)
        - ic_services (income, expense, net position)
        - capitalization (equity, debt, D/E ratio)
        - cash_pooling (identified, counterparty, positions)
        - priority_flags (array with priority, category, description, amount, source)
        - overall_tp_opportunity_score (A/B/C)
        - score_rationale
        - recommended_focus_areas
        - data_quality_notes
    </structure>
  </json_response_schema>
